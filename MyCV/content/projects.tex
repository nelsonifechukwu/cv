\sectiontitle{Selected Projects}

\entrytitlend{\href{https://www.nelsonelijah.com/projects/camai}{CamAI}: Action Recognition Camera System}{MediaPipe, Conv-LSTM, Computer Vision}{03/2023 - 05/2023}
\begin{details}
    \item Led a team of two to build an action-recognition system for patient monitoring in healthcare.
    \item Extracted pose features from the video feed with MediaPipe and classified patient state with a Conv-LSTM ensemble at 96\%+ accuracy, with reliable subject tracking, streamed to a web interface.
\end{details}

\entrytitlend{\href{https://www.nelsonelijah.com/projects/myquad}{Quad-X}: Quadcopter Flight Controller}{ATmega328-PU, PID, Sensor Fusion}{01/2021 - 12/2021}
\begin{details}
    \item Designed and built a quadcopter flight controller on an ATmega328-PU, fusing IMU data through a PID control loop for stable all-axis flight.
    \item Extended Quad-X into \href{https://www.nelsonelijah.com/projects/autouav}{AutoUAV}, adding autonomous navigation, in-flight object detection, and live camera streaming.
\end{details}
