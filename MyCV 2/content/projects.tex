\sectiontitle{Selected Projects}

\entrytitlend{\href{https://www.nelsonelijah.com/projects/camai}{CamAI}: Action-Recognition Camera System}{MediaPipe, ConvLSTM, Computer Vision}{03/2023 - 05/2023}
\begin{details}
    \item Led a two-person team in developing a patient-monitoring camera system that detects and classifies human actions from live video.
    \item Extracted pose landmarks with MediaPipe and trained a ConvLSTM ensemble that classified patient states with 96\%+ accuracy, maintained reliable subject tracking, and streamed results to a web interface.
\end{details}

\entrytitlend{\href{https://www.nelsonelijah.com/projects/myquad}{Quad-X}: Quadcopter Flight Controller}{ATmega328-PU, PID, Sensor Fusion}{01/2021 - 12/2021}
\begin{details}
    \item Designed and built an ATmega328-PU-based quadcopter flight controller, fusing IMU measurements within PID control loops to achieve stable attitude control across all axes.
    \item Extended the platform into \href{https://www.nelsonelijah.com/projects/autouav}{AutoUAV}, adding autonomous navigation, in-flight object detection, precision landing, and live camera streaming.
\end{details}
